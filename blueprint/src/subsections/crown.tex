\section{The crown theorems}\label{sec:crown}

\begin{theorem}[Engine, modular form]\label{thm:fltvandiver}
  \lean{FltVandiver.flt_vandiver}
  \leanok
  \uses{def:swcert, def:caseIholds, thm:nosituation}
  If \code{SmallWitnessCert}~$p$ holds and Case~I holds at $p$, then
  \code{FermatLastTheoremFor}~$p$: Case~II is closed by the descent, Case~I taken
  as hypothesis.
\end{theorem}

\begin{theorem}[Engine, discharged form]\label{thm:certs95}
  \lean{FltVandiver.Descent95.fermatLastTheoremFor_of_certs_95}
  \leanok
  \uses{thm:nosituation, def:swcert, thm:caseIb5}
  Given the $Q_i$ certificate, the size bound $\ell < p^2-p$, and the irregularity
  certificate with $p-3 \notin L$, \code{FermatLastTheoremFor}~$p$ holds with nothing
  about Case~I assumed --- the non-Wolstenholme headline.
\end{theorem}

\begin{theorem}[Reduction to two open problems]\label{thm:crown}
  \lean{FltVandiver.fermatLastTheorem_of_vandiver_sgAux}
  \leanok
  \uses{thm:fltvandiver, thm:caseIsg}
  If every prime has a small-witness certificate and every odd prime has a Sophie
  Germain auxiliary, then Fermat's Last Theorem holds --- with no modularity input.
  This is the paper's headline reduction; both hypotheses are classically open.
\end{theorem}

\begin{theorem}[Widened budget]\label{thm:sizecert}
  \lean{FltVandiver.BadCert.fermatLastTheoremFor_of_size_cert}
  \leanok
  \uses{thm:certs95, def:vcert, def:sgcert}
  The discharged form with the budget $\ell < p^2 - p$ replaced by
  $2\ell \le 3p^2 - 5p + 2$ and $p \nmid (\ell-1)/p$, with no other change: the bad set
  of Washington's Lemma~9.7 is proved empty in that range
  (\code{noBad\_of\_two\_mul\_le}). Case~I runs on a separate Sophie Germain prime.
\end{theorem}

\begin{theorem}[Bad-set certificate]\label{thm:badcert}
  \lean{FltVandiver.BadCert.fermatLastTheoremFor_of_bad_cert}
  \leanok
  \uses{thm:nosituation, def:vcert, def:sgcert}
  Case~II with no size bound on $\ell$: the $Q_i$ certificate, $p \nmid (\ell-1)/p$, and
  the computable emptiness certificate \code{noBadCert} for the bad set of Lemma~9.7,
  together with a Sophie Germain prime for Case~I, give
  \code{FermatLastTheoremFor}~$p$.
\end{theorem}

\begin{theorem}[Joint auxiliary]\label{thm:jointcert}
  \lean{FltVandiver.JointAux.fermatLastTheoremFor_of_joint_cert}
  \leanok
  \uses{thm:nosituation, def:vcert, def:sgcert, thm:caseIsg}
  One prime $\ell$ carrying both certificates, the all-even $Q_i$ test and the Sophie
  Germain conditions, with $p \nmid (\ell-1)/p$ and no upper bound on $\ell$, gives
  \code{FermatLastTheoremFor}~$p$: Legendre's condition (A) supplies the seed
  divisibility $\ell \mid z$ that the size bound used to buy.
\end{theorem}
